function frameStats(values) {
  assert.ok(values.length > 0 && values.every(n => Number.isFinite(n) && n > 0));
  const sorted = [...values].sort((a, b) => a - b);
  return { frames: values.length, fps: 1000 * values.length / values.reduce((a, b) => a + b, 0),
    p50Ms: sorted[Math.ceil(sorted.length * .5) - 1], p95Ms: sorted[Math.ceil(sorted.length * .95) - 1],
    p99Ms: sorted[Math.ceil(sorted.length * .99) - 1], maxMs: sorted.at(-1),
    over50Ms: values.filter(n => n > 50).length, over100Ms: values.filter(n => n > 100).length,
    over1000Ms: values.filter(n => n > 1000).length };
}
if (process.argv.includes('--self-test')) {
  assert.deepEqual(frameStats([10, 20, 30, 40, 100]), { frames: 5, fps: 25, p50Ms: 30, p95Ms: 100,
    p99Ms: 100, maxMs: 100, over50Ms: 1, over100Ms: 0, over1000Ms: 0 });
  assert.throws(() => frameStats([])); assert.throws(() => frameStats([NaN]));
  console.log('PASS E5 frame statistics self-check.'); process.exit(0);
}
const minutes = Number(process.env.CITY2127_SOAK_MINUTES ?? 30);
const compareSeconds = Number(process.env.CITY2127_COMPARE_SECONDS ?? 60);
assert.ok(Number.isFinite(minutes) && minutes >= 1);
assert.ok(Number.isFinite(compareSeconds) && compareSeconds >= 5);

  questionsPath: new URL('../survey/src/survey/questions.exhibition.json', import.meta.url).pathname });
const { server, realtime } = createSurveyServer({ ctx, staticDir: new URL('../survey/dist', import.meta.url).pathname });
let browser;
const report = { startedAt: new Date().toISOString(), checkedCommit: process.env.CITY2127_CHECKED_COMMIT ?? 'working tree',
  runtimeSourceHashes: Object.fromEntries(['src/main.ts', 'src/energyRuntime.ts', 'src/energyPrototype.ts', 'tests/energyE5.browser.mjs'].map(p =>

  mode: 'Vite DEV; authored hero; headed Chrome; real clock; scratch SQLite; loopback only',
  viewport: { width: 1920, height: 1080 }, deviceScaleFactor: 1, phases: [], proposals: [], errors: [], warnings: [],

try {

  const cityOrigin = process.env.CITY2127_CITY_URL ?? 'http://127.0.0.1:5173';
  const citySockets = new Set(); let cityConnections = 0;
  server.on('upgrade', (request, socket) => {
    if (request.headers.origin !== new URL(cityOrigin).origin) return;
    cityConnections++; citySockets.add(socket); socket.on('close', () => citySockets.delete(socket));
  });
  browser = await chromium.launch({ channel: 'chrome', headless: false });
  report.browser = browser.version();

  const gpu = (await browserCDP.send('SystemInfo.getInfo')).gpu;
  report.gpu = { devices: gpu.devices, renderer: gpu.auxAttributes.glRenderer, features: gpu.featureStatus };
  report.host = execFileSync('system_profiler', ['SPHardwareDataType', 'SPSoftwareDataType'], { encoding: 'utf8' })
    .split('\n').filter(line => !/Serial Number|UUID|UDID|User Name|Boot Volume/.test(line)).join('\n');
  report.node = process.version;
  const display = await browser.newContext({ viewport: report.viewport, deviceScaleFactor: 1 });
  await display.addInitScript(() => {
    let last, frames = [], losses = 0;
    document.addEventListener('webglcontextlost', () => losses++, true);
    const tick = t => { if (last !== undefined) frames.push(t - last); last = t; requestAnimationFrame(tick); };
    requestAnimationFrame(tick);
    window.e5Frames = () => { const result = { frames, losses, visibility: document.visibilityState }; frames = []; return result; };
  });
  const city = await display.newPage();
  const a = await browser.newPage({ viewport: { width: 1280, height: 900 } });
  const b = await browser.newPage({ viewport: { width: 1280, height: 900 } });
  for (const [name, page] of [['city', city], ['A', a], ['B', b]]) {

    page.on('crash', () => report.errors.push({ name, error: 'page crashed' }));

    page.on('response', r => { if (r.status() >= 400) report.badResponses.push({ name, url: r.url(), status: r.status() }); });
    page.on('requestfailed', r => report.failedRequests.push({ name, url: r.url(), error: r.failure()?.errorText }));
  }
  const cdp = await display.newCDPSession(city); await cdp.send('Performance.enable');
  const post = async (path, data) => { const r = await display.request.post(survey + path, { data }); const body = await r.json(); assert.ok(r.ok() && body.ok, JSON.stringify(body)); return body.data; };
  const view = async () => (await (await display.request.get(survey + '/api/city-view')).json()).data;
  const url = (quality, hour) => `${cityOrigin}/?survey=${encodeURIComponent(survey.replace('http', 'ws') + '/ws')}&quality=${quality}${hour === undefined ? '' : `&hour=${hour}`}`;
  async function load(quality, hour) {
    await city.goto(url(quality, hour)); await city.bringToFront();
    await city.waitForFunction(() => Number(document.querySelector('canvas')?.dataset.triangles) > 0, null, { timeout: 120_000 });

    await city.evaluate(() => window.e5Frames());
  }
  async function sample(phase, start) {
    const frames = await city.evaluate(() => window.e5Frames());
    assert.equal(frames.visibility, 'visible'); assert.equal(frames.losses, 0); assert.ok(frames.frames.length);
    const canvas = await city.locator('canvas').evaluate(c => ({ css: [c.clientWidth, c.clientHeight], buffer: [c.width, c.height],
      fps: Number(c.dataset.fps), calls: Number(c.dataset.drawCalls), triangles: Number(c.dataset.triangles),
      geometries: Number(c.dataset.geometries), textures: Number(c.dataset.textures), hour: Number(c.dataset.hour),
      actorTime: Number(c.dataset.time), energy: JSON.parse(c.dataset.energy), presentation: document.querySelector('.causal-panel')?.dataset.presentation }));
    assert.deepEqual(canvas.css, [1920, 1080]); assert.deepEqual(canvas.buffer, [1920, 1080]);
    assert.ok(canvas.triangles > 0 && canvas.calls > 0);
    const metrics = Object.fromEntries((await cdp.send('Performance.getMetrics')).metrics.map(m => [m.name, m.value]));
    const row = { phase, elapsedSeconds: (performance.now() - start) / 1000, at: new Date().toISOString(),
      ...frameStats(frames.frames), ...canvas, jsHeapUsed: metrics.JSHeapUsedSize, jsHeapTotal: metrics.JSHeapTotalSize,
      nodes: metrics.Nodes, serverRss: process.memoryUsage().rss, contextLosses: frames.losses, visibility: frames.visibility };
    appendFileSync(join(evidence, 'samples.ndjson'), JSON.stringify(row) + '\n');
    return { row, intervals: frames.frames };
  }
  async function phase(name, seconds, work) {
    const start = performance.now(), intervals = [], rows = []; let nextWork = 0, nextProgress = 0;
    await city.evaluate(() => window.e5Frames());
    while (performance.now() - start < seconds * 1000) {
      if (work && performance.now() - start >= nextWork) { await work(); nextWork += 40_000; await city.bringToFront(); }
      await delay(5_000);
      const result = await sample(name, start); rows.push(result.row); intervals.push(...result.intervals);

    }
    const result = { name, seconds: (performance.now() - start) / 1000, ...frameStats(intervals), samples: rows.length,
      first: rows[0], last: rows.at(-1), minFps: Math.min(...rows.map(r => r.fps)), maxFps: Math.max(...rows.map(r => r.fps)) };
